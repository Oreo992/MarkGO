import { createHmac, timingSafeEqual } from "node:crypto";

function secret(): string {
  return process.env.SIKU_SECRET ?? "dev-secret-change-me";
}

export function checkPassword(input: string): boolean {
  const expected = process.env.SIKU_PASSWORD ?? "";
  if (!expected || !input) return false;
  const a = Buffer.from(input);
  const b = Buffer.from(expected);
  return a.length === b.length && timingSafeEqual(a, b);
}

export function issueToken(): string {
  const payload = String(Date.now());
  const sig = createHmac("sha256", secret()).update(payload).digest("hex");
  return `${payload}.${sig}`;
}

export function verifyToken(token: string | undefined): boolean {
  if (!token) return false;
  const dot = token.lastIndexOf(".");
  if (dot <= 0) return false;
  const payload = token.slice(0, dot);
  const sig = token.slice(dot + 1);
  const expect = createHmac("sha256", secret()).update(payload).digest("hex");
  const a = Buffer.from(sig);
  const b = Buffer.from(expect);
  return a.length === b.length && timingSafeEqual(a, b);
}
```

import { mkdir, writeFile, readdir, readFile, access } from "node:fs/promises";
import { join } from "node:path";
import { execFile } from "node:child_process";
import { promisify } from "node:util";

const run = promisify(execFile);

export interface CaptureInput {
  title?: string;
  text?: string;
  url?: string;
}

export interface PageMeta {
  file: string;
  title: string;
  type: string;
}

const SECTIONS = ["sources", "entities", "concepts", "thinking"] as const;
export type Section = (typeof SECTIONS)[number];

export class Vault {
  constructor(readonly root: string) {}

  get inboxDir(): string {
    return join(this.root, ".inbox");
  }
  get wikiDir(): string {
    return join(this.root, "wiki");
  }

  async capture(input: CaptureInput): Promise<string> {
    if (!input.text && !input.url) throw new Error("捕获内容为空");
    await mkdir(this.inboxDir, { recursive: true });
    const stamp = new Date().toISOString().replace(/[:.]/g, "-");
    const slug = (input.title ?? "capture").replace(/[^\p{L}\p{N}-]+/gu, "-").slice(0, 40) || "capture";
    const file = join(this.inboxDir, `${stamp}-${slug}.md`);
    const lines = [
      "---",
      "type: capture",
      `title: ${JSON.stringify(input.title ?? slug)}`,
      ...(input.url ? [`resource: ${input.url}`] : []),
      `timestamp: ${new Date().toISOString()}`,
      "---",
      "",
      input.text ?? `来源链接：${input.url}`,
      "",
    ];
    await writeFile(file, lines.join("\n"), "utf8");
    return file;
  }

  async listInbox(): Promise<string[]> {
    await mkdir(this.inboxDir, { recursive: true });
    const names = await readdir(this.inboxDir);
    return names.filter((n) => n.endsWith(".md")).sort().map((n) => join(this.inboxDir, n));
  }

  async listPages(section: Section): Promise<PageMeta[]> {
    const dir = join(this.wikiDir, section);
    let names: string[];
    try {
      names = await readdir(dir);
    } catch {
      return [];
    }
    const pages: PageMeta[] = [];
    for (const n of names.filter((n) => n.endsWith(".md")).sort()) {
      const raw = await readFile(join(dir, n), "utf8");
      pages.push({
        file: `${section}/${n}`,
        title: fmValue(raw, "title") ?? n.replace(/\.md$/, ""),
        type: fmValue(raw, "type") ?? section,
      });
    }
    return pages;
  }

  async readPage(rel: string): Promise<string> {
    if (rel.includes("..") || rel.startsWith("/")) throw new Error("非法路径");
    return readFile(join(this.wikiDir, rel), "utf8");
  }

  async resolvePage(name: string): Promise<string | null> {
    for (const section of SECTIONS) {
      const rel = `${section}/${name}.md`;
      try {
        await access(join(this.wikiDir, rel));
        return rel;
      } catch {
        // 该分区没有，继续找下一个
      }
    }
    return null;
  }

  async commit(message: string): Promise<void> {
    await run("git", ["add", "-A"], { cwd: this.root });
    const { stdout } = await run("git", ["status", "--porcelain"], { cwd: this.root });
    if (stdout.trim() === "") return; // 无变更，跳过
    await run("git", ["commit", "-m", message], { cwd: this.root });
  }
}

function fmValue(raw: string, key: string): string | undefined {
  const m = raw.match(new RegExp(`^${key}:\\s*(.+)$`, "m"));
  return m?.[1]?.trim().replace(/^"(.*)"$/, "$1");
}
```

import { appendFile, readFile, mkdir } from "node:fs/promises";
import { dirname } from "node:path";

export interface FeedEvent {
  id: string;
  kind: "captured" | "ingested" | "failed";
  title: string;
  detail: string;
  at: string;
}

export class Feed {
  constructor(private file: string) {}

  async append(ev: FeedEvent): Promise<void> {
    await mkdir(dirname(this.file), { recursive: true });
    await appendFile(this.file, JSON.stringify(ev) + "\n", "utf8");
  }

  async recent(limit = 50): Promise<FeedEvent[]> {
    let raw: string;
    try {
      raw = await readFile(this.file, "utf8");
    } catch {
      return [];
    }
    return raw
      .trim()
      .split("\n")
      .filter(Boolean)
      .map((l) => JSON.parse(l) as FeedEvent)
      .slice(-limit)
      .reverse();
  }
}

import { basename, join } from "node:path";
import { rename, mkdir } from "node:fs/promises";
import { randomUUID } from "node:crypto";
import type { Vault } from "./vault.js";
import type { Feed } from "./feed.js";
import type { AgentRunner } from "./agent-types.js";

// 摄入斜杠命令以 docs/SPIKE-NOTES.md 验证结果为准；默认按 claude-obsidian README 的 /wiki-ingest。
export function ingestPrompt(file: string): string {
  return `/wiki-ingest ${file}`;
}

export class IngestQueue {
  private chain: Promise<void> = Promise.resolve();

  constructor(
    private vault: Vault,
    private feed: Feed,
    private agent: AgentRunner
  ) {}

  async start(): Promise<void> {
    for (const file of await this.vault.listInbox()) this.enqueue(file);
  }

  enqueue(file: string): void {
    this.chain = this.chain.then(() => this.process(file));
  }

  idle(): Promise<void> {
    return this.chain;
  }

  private async process(file: string): Promise<void> {
    const title = basename(file);
    try {
      const summary = await this.agent.run(ingestPrompt(file), process.env.SIKU_INGEST_MODEL);
      const rawDir = join(this.vault.root, ".raw");
      await mkdir(rawDir, { recursive: true });
      await rename(file, join(rawDir, title));
      await this.vault.commit(`ingest: ${title}`);
      await this.feed.append({
        id: randomUUID(),
        kind: "ingested",
        title,
        detail: summary.slice(0, 500),
        at: new Date().toISOString(),
      });
    } catch (err) {
      await this.feed.append({
        id: randomUUID(),
        kind: "failed",
        title,
        detail: String(err),
        at: new Date().toISOString(),
      });
    }
  }
}

import { query } from "@anthropic-ai/claude-agent-sdk";
import type { AgentRunner } from "./agent-types.js";

export function createAgentRunner(vaultRoot: string): AgentRunner {
  return {
    async run(prompt: string, model?: string): Promise<string> {
      let out = "";
      for await (const msg of query({
        prompt,
        options: {
          cwd: vaultRoot,
          permissionMode: "bypassPermissions",
          settingSources: ["project"],
          ...(model ? { model } : {}),
        },
      })) {
        if (msg.type === "result") {
          if (msg.subtype === "success") out = msg.result;
          else throw new Error(`Agent 运行失败: ${msg.subtype}`);
        }
      }
      return out;
    },
  };
}

import express, { type Express } from "express";
import cookieParser from "cookie-parser";
import { randomUUID } from "node:crypto";
import type { Vault, Section } from "./vault.js";
import type { Feed } from "./feed.js";
import type { IngestQueue } from "./queue.js";
import type { AgentRunner } from "./agent-types.js";
import { queryPrompt } from "./agent.js";
import { checkPassword, issueToken, verifyToken } from "./auth.js";

export interface AppDeps {
  vault: Vault;
  feed: Feed;
  queue: IngestQueue;
  agent: AgentRunner;
}

export function createApp({ vault, feed, queue, agent }: AppDeps): Express {
  const app = express();
  app.use(express.json({ limit: "10mb" }));
  app.use(cookieParser());

  app.post("/api/login", (req, res) => {
    if (checkPassword(String(req.body?.password ?? ""))) {
      res.cookie("siku", issueToken(), {
        httpOnly: true,
        sameSite: "lax",
        maxAge: 30 * 24 * 3600 * 1000,
      });
      res.json({ ok: true });
    } else {
      res.status(401).json({ error: "密码错误" });
    }
  });

  app.use("/api", (req, res, next) => {
    if (verifyToken(req.cookies?.siku)) next();
    else res.status(401).json({ error: "未登录" });
  });

  app.post("/api/capture", async (req, res) => {
    try {
      const file = await vault.capture({
        title: req.body?.title,
        text: req.body?.text,
        url: req.body?.url,
      });
      await feed.append({
        id: randomUUID(),
        kind: "captured",
        title: req.body?.title ?? "未命名",
        detail: "已进入摄入队列",
        at: new Date().toISOString(),
      });
      queue.enqueue(file);
      res.json({ ok: true });
    } catch {
      res.status(400).json({ error: "捕获内容为空" });
    }
  });

  app.get("/api/feed", async (_req, res) => {
    res.json(await feed.recent());
  });

  app.post("/api/ask", async (req, res) => {
    const question = String(req.body?.question ?? "").trim();
    if (!question) {
      res.status(400).json({ error: "问题为空" });
      return;
    }
    const answer = await agent.run(queryPrompt(question));
    res.json({ answer });
  });

  app.get("/api/pages/:section", async (req, res) => {
    res.json(await vault.listPages(req.params.section as Section));
  });

  app.get("/api/page", async (req, res) => {
    try {
      res.type("text/markdown").send(await vault.readPage(String(req.query.path ?? "")));
    } catch {
      res.status(404).json({ error: "页面不存在" });
    }
  });

  app.get("/api/resolve", async (req, res) => {
    res.json({ path: await vault.resolvePage(String(req.query.name ?? "")) });
  });

  return app;
}

export async function api<T>(path: string, body?: unknown): Promise<T> {
  const res = await fetch(
    `/api/${path}`,
    body === undefined
      ? { credentials: "include" }
      : {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify(body),
          credentials: "include",
        }
  );
  if (res.status === 401) {
    location.hash = "#/login";
    throw new Error("未登录");
  }
  if (!res.ok) throw new Error(`请求失败：${res.status}`);
  const ct = res.headers.get("content-type") ?? "";
  return (ct.includes("json") ? res.json() : res.text()) as Promise<T>;
}

export function el(html: string): HTMLElement {
  const t = document.createElement("template");
  t.innerHTML = html.trim();
  return t.content.firstElementChild as HTMLElement;
}

        <textarea name="text" placeholder="粘贴任何内容——文章、AI 对话、想法……" rows="6"></textarea>
        <input name="url" placeholder="或者贴一个链接" />
        <button type="submit">扔进去</button>
        <p class="hint" hidden></p>
      </form>
      <h2>摄入流水</h2>
      <ul class="feed"></ul>
    </section>
  `);
  const form = view.querySelector("form")!;
  const hint = view.querySelector(".hint") as HTMLElement;
  const list = view.querySelector(".feed")!;

  form.addEventListener("submit", async (e) => {
    e.preventDefault();
    const data = new FormData(form);
    const text = String(data.get("text") ?? "").trim();
    const url = String(data.get("url") ?? "").trim();
    if (!text && !url) {
      hint.textContent = "先粘贴点内容或链接";
      hint.hidden = false;
      return;
    }
    await api("capture", {
      title: String(data.get("title") ?? "").trim() || undefined,
      text: text || undefined,
      url: url || undefined,
    });
    form.reset();
    hint.textContent = "已扔进去，Agent 正在消化";
    hint.hidden = false;
    await refresh();
  });

  async function refresh(): Promise<void> {
    const events = await api<FeedEvent[]>("feed");
    list.replaceChildren(
      ...events.map((ev) =>
        el(`
          <li class="${ev.kind}">
            <span class="kind">${KIND_LABEL[ev.kind]}</span>
            <span class="title">${ev.title}</span>
            <p class="detail">${ev.detail}</p>
            <time>${new Date(ev.at).toLocaleString("zh-CN")}</time>
          </li>
        `)
      )
    );
  }

  void refresh();
  const timer = setInterval(refresh, 5000);
  window.addEventListener("hashchange", () => clearInterval(timer), { once: true });
  root.append(view);
}

export function renderWikilinks(md: string): string {
  return md.replace(/\[\[([^\]|]+)(?:\|([^\]]+))?\]\]/g, (_m, target: string, label?: string) => {
    const name = target.trim();
    return `<a class="wikilink" href="#/library/page/${encodeURIComponent(name)}">${label?.trim() ?? name}</a>`;
  });
}

```ts
import { marked } from "marked";
import { api, el } from "../api.js";
import { renderWikilinks } from "../wikilink.js";

interface PageMeta {
  file: string;
  title: string;
  type: string;
}

const SECTIONS: { key: string; label: string }[] = [
  { key: "sources", label: "来源" },
  { key: "entities", label: "实体" },
  { key: "concepts", label: "概念" },
  { key: "thinking", label: "思考产出" },
];

export function renderLibrary(root: HTMLElement, pageName?: string): void {
  if (pageName) {
    void renderPage(root, pageName);
    return;
  }
  const view = el(`
    <section class="library">
      <form class="ask">
        <input placeholder="问你的知识库——比如：闭环怎么打？" />
        <button type="submit">问</button>
      </form>
      <article class="answer" hidden></article>
      <div class="sections"></div>
    </section>
  `);
  const form = view.querySelector("form")!;
  const input = view.querySelector("input")!;
  const answer = view.querySelector(".answer") as HTMLElement;
  const sections = view.querySelector(".sections")!;

  form.addEventListener("submit", async (e) => {
    e.preventDefault();
    const question = input.value.trim();
    if (!question) return;
    answer.hidden = false;
    answer.textContent = "Agent 正在翻阅你的库……";
    try {
      const { answer: text } = await api<{ answer: string }>("ask", { question });
      answer.innerHTML = renderWikilinks(await marked.parse(text));
    } catch (err) {
      answer.textContent = `回答失败：${String(err)}`;
    }
  });

  void (async () => {
    for (const s of SECTIONS) {
      const pages = await api<PageMeta[]>(`pages/${s.key}`);
      if (pages.length === 0) continue;
      const block = el(`<div class="section"><h2>${s.label}（${pages.length}）</h2><ul></ul></div>`);
      block.querySelector("ul")!.replaceChildren(
        ...pages.map((p) =>
          el(`<li><a href="#/library/page/${encodeURIComponent(p.file)}">${p.title}</a></li>`)
        )
      );
      sections.append(block);
    }
  })();

  root.append(view);
}

async function renderPage(root: HTMLElement, name: string): Promise<void> {
  const view = el(`<article class="page"><p>加载中……</p></article>`);
  root.append(view);
  // name 可能是 "concepts/x.md" 相对路径，也可能是 wikilink 的裸名——后者先 resolve
  let path = name.includes("/") ? name : null;
  if (!path) {
    const r = await api<{ path: string | null }>(`resolve?name=${encodeURIComponent(name)}`);
    path = r.path;
  }
  if (!path) {
    view.innerHTML = `<p>库里还没有「${name}」这一页。</p><p><a href="#/library">← 回图书馆</a></p>`;
    return;
  }
  const md = await api<string>(`page?path=${encodeURIComponent(path)}`);
  view.innerHTML =
    `<p><a href="#/library">← 回图书馆</a></p>` + renderWikilinks(await marked.parse(md));
}
```
